Private Sub SetInsideSize(ByVal w As Single, ByVal h As Single)

    Me.Height = Me.Height + (h - Me.InsideHeight)
End Sub

' O'zbekcha apostroflar: o' g' -> o‘ g‘, qolgani -> ’
Private Function Uz(ByVal s As String) As String
    s = Replace(s, "o'", "o" & ChrW(&H2018))
    s = Replace(s, "O'", "O" & ChrW(&H2018))
    s = Replace(s, "g'", "g" & ChrW(&H2018))
    s = Replace(s, "G'", "G" & ChrW(&H2018))
    Uz = Replace(s, "'", ChrW(&H2019))
End Function

Private Function TextTop(ByVal rowTop As Single, ByVal size As Single) As Single

Private Function CardHeight(ByVal rows As Long, ByVal extra As Single) As Single

' Yumaloq burchakli to'rtburchak: rangli Label + 4 ta burchak rasmi
Private Function RoundBox(ByVal l As Single, ByVal t As Single, ByVal w As Single, ByVal h As Single, _
        ByVal fill As Long, ByVal bg As Long, ByVal r As Single) As MSForms.Label
    Dim pos As Long, img As MSForms.Image
    Set RoundBox = Me.Controls.Add("Forms.Label.1")
    With RoundBox
        .Caption = "": .Left = l: .Top = t: .Width = w: .Height = h
        .BackStyle = fmBackStyleOpaque: .BackColor = fill
        .BorderStyle = fmBorderStyleNone
    End With
    For pos = 0 To 3
        Set img = Me.Controls.Add("Forms.Image.1")
        With img
            .BorderStyle = fmBorderStyleNone
            .BackColor = bg
            .PictureSizeMode = fmPictureSizeModeStretch
            .Picture = LoadPicture(CornerFile(r, fill, bg, pos))
            .Width = r: .Height = r
            .Left = IIf(pos = 0 Or pos = 2, l, l + w - r)
            .Top = IIf(pos < 2, t, t + h - r)
        End With
    Next pos
End Function

' Burchak rasmi (BMP) - TEMP papkada bir marta yaratiladi
' pos: 0 = chap-yuqori, 1 = o'ng-yuqori, 2 = chap-past, 3 = o'ng-past
Private Function CornerFile(ByVal rPt As Single, ByVal inner As Long, ByVal outer As Long, _
        ByVal pos As Long) As String
    Dim n As Long, f As String, fh As Integer, rowLen As Long, dataLen As Long
    Dim b() As Byte, x As Long, y As Long, sx As Long, sy As Long, cnt As Long, i As Long
    Dim cx As Double, cy As Double, dx As Double, dy As Double, a As Double

    n = CLng(rPt * 4 / 3)
    If n < 2 Then n = 2
    f = Environ$("TEMP") & "\pechat_c" & n & "_" & Hex$(inner) & "_" & Hex$(outer) & "_" & pos & ".bmp"
    CornerFile = f
    If Len(Dir$(f)) > 0 Then Exit Function

    rowLen = (n * 3 + 3) And Not 3
    dataLen = rowLen * n
    ReDim b(0 To 53 + dataLen)
    b(0) = 66: b(1) = 77                 ' "BM"
    PutLong b, 2, 54 + dataLen
    PutLong b, 10, 54
    PutLong b, 14, 40
    PutLong b, 18, n
    PutLong b, 22, n
    b(26) = 1: b(28) = 24                ' 1 tekislik, 24 bit
    PutLong b, 34, dataLen
    PutLong b, 38, 2835: PutLong b, 42, 2835

    cx = IIf(pos = 0 Or pos = 2, n, 0)
    cy = IIf(pos < 2, n, 0)
    For y = 0 To n - 1
        For x = 0 To n - 1
            cnt = 0
            For sy = 0 To 3
                For sx = 0 To 3
                    dx = x + (sx + 0.5) / 4 - cx
                    dy = y + (sy + 0.5) / 4 - cy
                    If dx * dx + dy * dy <= n * n Then cnt = cnt + 1
                Next sx
            Next sy
            a = cnt / 16
            i = 54 + (n - 1 - y) * rowLen + x * 3
            b(i) = Mix((inner \ 65536) And 255, (outer \ 65536) And 255, a)
            b(i + 1) = Mix((inner \ 256) And 255, (outer \ 256) And 255, a)
            b(i + 2) = Mix(inner And 255, outer And 255, a)
        Next x
    Next y

    fh = FreeFile
    Open f For Binary Access Write As #fh
    Put #fh, , b
    Close #fh
End Function

Private Sub PutLong(b() As Byte, ByVal offset As Long, ByVal v As Long)
    b(offset) = v And 255
    b(offset + 1) = (v \ 256) And 255
    b(offset + 2) = (v \ 65536) And 255
    b(offset + 3) = (v \ 16777216) And 255
End Sub

Private Function Mix(ByVal c1 As Long, ByVal c0 As Long, ByVal a As Double) As Byte
    Mix = CByte(c0 + (c1 - c0) * a)
End Function

Private Sub SelectItem(ByVal cbo As MSForms.ComboBox, ByVal txt As String, ByVal defIndex As Long)
    Dim i As Long
    For i = 0 To cbo.ListCount - 1
        If StrComp(cbo.List(i), txt, vbTextCompare) = 0 Then cbo.ListIndex = i: Exit Sub
    Next i
    cbo.ListIndex = defIndex
End Sub

Private Sub ShowMsg(ByVal txt As String, ByVal color As Long, Optional ByVal tip As String = "")
    lblMsg.Caption = Uz(txt)
    lblMsg.ForeColor = color
    lblMsg.ControlTipText = tip
End Sub

'------------------------------------------------------------------------------
'  Sozlamalarni eslab qolish
'------------------------------------------------------------------------------
Private Sub LoadSettings()
    Dim folder As String
    SelectItem cboColor, GetSetting(APP_NAME, "Last", "Color", "CMYK"), 0
    SelectItem cboType, GetSetting(APP_NAME, "Last", "Type", "Banner"), 0
    SelectItem cboFormat, GetSetting(APP_NAME, "Last", "Format", "TIF"), 0
    brightVal = CLng(Val(GetSetting(APP_NAME, "Last", "Bright", "15")))
    If brightVal < 0 Or brightVal > 50 Then brightVal = 15
    txtDpi.Text = GetSetting(APP_NAME, "Last", "Dpi", "150")

    folder = ActiveDocument.FilePath
    If Len(folder) = 0 Then folder = GetSetting(APP_NAME, "Last", "Folder", "")
    If Len(folder) = 0 Then folder = CreateObject("WScript.Shell").SpecialFolders("Desktop")
    txtFolder.Text = folder
End Sub

Private Sub SaveSettings()
    SaveSetting APP_NAME, "Last", "Color", cboColor.Text
    SaveSetting APP_NAME, "Last", "Bright", CStr(brightVal)
    SaveSetting APP_NAME, "Last", "Type", cboType.Text
    SaveSetting APP_NAME, "Last", "Format", cboFormat.Text
    SaveSetting APP_NAME, "Last", "Dpi", txtDpi.Text
    SaveSetting APP_NAME, "Last", "Folder", txtFolder.Text
End Sub

Private Function IsCMYK() As Boolean
    IsCMYK = (cboColor.Text = "CMYK")
End Function

Private Function NameFor(ByVal sr As ShapeRange) As String
    Dim x As Double, y As Double, w As Double, h As Double
    GetBoundsCm sr, x, y, w, h
    NameFor = BuildFileName(w, h, cboType.Text, IsCMYK(), cboFormat.Text)
End Function

Private Sub UpdateUI()
    Dim i As Long, txt As String, one As ShapeRange
    If loading Then Exit Sub

    ' Ochroqlik
    If brightVal = 0 Then
        lblBrVal.Caption = Uz("Yo'q")
        lblBrUnit.Visible = False
    Else
        lblBrVal.Caption = CStr(brightVal)
        lblBrUnit.Visible = True
    End If

    ' PNG faqat RGB bo'ladi
    If cboFormat.Text = "PNG" Then
        If cboColor.Text <> "RGB" Then cboColor.ListIndex = 1
        cboColor.Enabled = False
        ShowMsg "PNG faqat RGB", CLR_WARN, Uz("PNG formati CMYK ni qo'llamaydi")
    Else
        cboColor.Enabled = True
        ShowMsg "", CLR_MUTED
    End If

    ' Nima saqlanadi
    If sel.Count = 0 Then
        lblCount.Caption = "Sahifa"
        lblCount.ControlTipText = Uz("Hech narsa tanlanmagan: butun sahifa saqlanadi")
        txt = NameFor(Nothing)
    Else
        lblCount.Caption = sel.Count & " ta obyekt"
        lblCount.ControlTipText = Uz("Har bir obyekt alohida faylga saqlanadi")
        For i = 1 To sel.Count
            If i > 3 Then
                txt = txt & "+" & (sel.Count - 3) & " ta"
                Exit For
            End If
            Set one = CreateShapeRange
            one.Add sel(i)
            txt = txt & NameFor(one) & vbCrLf
        Next i
    End If
    lblPreview.Caption = txt
End Sub

Private Sub StepBright(ByVal delta As Long)
    brightVal = brightVal + delta * 5
    If brightVal < 0 Then brightVal = 0
    If brightVal > 50 Then brightVal = 50
    UpdateUI
End Sub

Private Sub StepDpi(ByVal delta As Long)
    Dim p As Variant, i As Long, cur As Long
    p = Split(DPI_PRESETS, ",")
    cur = CLng(Val(txtDpi.Text))
    If delta > 0 Then
        For i = 0 To UBound(p)
            If CLng(p(i)) > cur Then txtDpi.Text = p(i): Exit Sub
        Next i
    Else
        For i = UBound(p) To 0 Step -1
            If CLng(p(i)) < cur Then txtDpi.Text = p(i): Exit Sub
        Next i
    End If
End Sub

Private Sub PickFolder()
    Dim folder As String
    folder = CorelScriptTools.GetFolder(txtFolder.Text, "Papka")
    If Len(folder) > 0 Then txtFolder.Text = folder
End Sub

Private Sub DoExport()
    Dim dpi As Long, result As String, n As Long, ln As Variant
    If busy Then Exit Sub

    dpi = CLng(Val(txtDpi.Text))
    If Not IsNumeric(txtDpi.Text) Or dpi < 10 Or dpi > 2400 Then
        ShowMsg "Razreshenie: 10" & ChrW(&H2013) & "2400 dpi", CLR_ERROR
        Exit Sub
    End If
    If Len(txtFolder.Text) = 0 Or Len(Dir$(txtFolder.Text, vbDirectory)) = 0 Then
        ShowMsg "Papka topilmadi", CLR_ERROR, txtFolder.Text
        Exit Sub
    End If

    busy = True
    SaveSettings
    ShowMsg "Eksport" & ChrW(&H2026), CLR_MUTED
    Me.Repaint

    result = ExportPechat(IsCMYK(), brightVal, cboType.Text, cboFormat.Text, dpi, txtFolder.Text)
    If Len(result) > 0 Then
        For Each ln In Split(result, vbCrLf)
            If Len(ln) > 0 Then n = n + 1
        Next ln
        ShowMsg "Saqlandi: " & n & " ta fayl", CLR_OK, result
    Else
        ShowMsg "Eksport bajarilmadi", CLR_ERROR
    End If
    busy = False
End Sub
