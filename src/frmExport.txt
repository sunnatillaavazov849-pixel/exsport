    Set optBanner = AddOpt(f, "Banner", 10, 6, 80)

    With lblPreview: .Height = 18: .Font.Bold = True: .ForeColor = RGB(0, 90, 170): End With

'------------------------------------------------------------------------------
'  Sozlamalarni eslab qolish
'------------------------------------------------------------------------------
Private Sub LoadSettings()
    Dim folder As String
    optCMYK.Value = (GetSetting(APP_NAME, "Last", "Color", "cmyk") = "cmyk")
    optRGB.Value = Not optCMYK.Value
    chkBright.Value = (GetSetting(APP_NAME, "Last", "Bright", "1") = "1")
    cboBright.Text = GetSetting(APP_NAME, "Last", "BrightPct", "15")

    Select Case GetSetting(APP_NAME, "Last", "Type", "banner")
        Case "orakal": optOrakal.Value = True
        Case "bekprint": optBekprint.Value = True
        Case Else: optBanner.Value = True
    End Select

    Select Case GetSetting(APP_NAME, "Last", "Format", "tif")
        Case "png": optPNG.Value = True
        Case "jpg": optJPG.Value = True
        Case "jpeg": optJPEG.Value = True
        Case Else: optTIF.Value = True
    End Select

    cboDpi.Text = GetSetting(APP_NAME, "Last", "Dpi", "150")

    optPage.Value = True
    If ActiveSelectionRange.Count = 0 Then optSel.Enabled = False

    folder = ""
    If Len(ActiveDocument.FilePath) > 0 Then folder = ActiveDocument.FilePath
    If Len(folder) = 0 Then folder = GetSetting(APP_NAME, "Last", "Folder", "")
    If Len(folder) = 0 Then folder = CreateObject("WScript.Shell").SpecialFolders("Desktop")
    txtFolder.Text = folder
End Sub

Private Sub SaveSettings()
    SaveSetting APP_NAME, "Last", "Color", IIf(optCMYK.Value, "cmyk", "rgb")
    SaveSetting APP_NAME, "Last", "Bright", IIf(chkBright.Value, "1", "0")
    SaveSetting APP_NAME, "Last", "BrightPct", cboBright.Text
    SaveSetting APP_NAME, "Last", "Type", SelectedType()
    SaveSetting APP_NAME, "Last", "Format", SelectedExt()
    SaveSetting APP_NAME, "Last", "Dpi", cboDpi.Text
    SaveSetting APP_NAME, "Last", "Folder", txtFolder.Text
End Sub

'------------------------------------------------------------------------------
'  Tanlovlar
'------------------------------------------------------------------------------
Private Function SelectedType() As String
    If optOrakal.Value Then
        SelectedType = "orakal"
    ElseIf optBekprint.Value Then
        SelectedType = "bekprint"
    Else
        SelectedType = "banner"
    End If
End Function

Private Function SelectedExt() As String
    If optPNG.Value Then
        SelectedExt = "png"
    ElseIf optJPG.Value Then
        SelectedExt = "jpg"
    ElseIf optJPEG.Value Then
        SelectedExt = "jpeg"
    Else
        SelectedExt = "tif"
    End If
End Function

Private Sub UpdateUI()
    Dim x As Double, y As Double, w As Double, h As Double
    If loading Then Exit Sub

    ' PNG faqat RGB bo'ladi
    If optPNG.Value Then
        optRGB.Value = True
        optCMYK.Enabled = False
        lblNote.Caption = "PNG formati CMYK ni qo'llamaydi - RGB bo'lib saqlanadi."
    Else
        optCMYK.Enabled = True
        lblNote.Caption = ""
    End If

    cboBright.Enabled = chkBright.Value

    If GetExportBounds(optSel.Value, x, y, w, h) Then
        lblPreview.Caption = BuildFileName(w, h, SelectedType(), optCMYK.Value, SelectedExt())
        If IsNumeric(cboDpi.Text) Then
            If Val(cboDpi.Text) > 0 Then
                lblNote.Caption = Trim$(lblNote.Caption & "  Piksel: " & _
                    CLng(w / 2.54 * Val(cboDpi.Text)) & " x " & CLng(h / 2.54 * Val(cboDpi.Text)))
            End If
        End If
    Else
        lblPreview.Caption = "(obyekt tanlanmagan)"
    End If
End Sub

Private Sub btnFolder_Click()
    Dim folder As String
    folder = CorelScriptTools.GetFolder(txtFolder.Text, "Eksport papkasini tanlang")
    If Len(folder) > 0 Then txtFolder.Text = folder
End Sub

Private Sub btnCancel_Click()
    Unload Me
End Sub

Private Sub btnExport_Click()
    Dim dpi As Long, pct As Long, result As String

    If Not IsNumeric(cboDpi.Text) Then
        MsgBox "Razreshenie (dpi) raqam bo'lishi kerak.", vbExclamation, "Pechat eksport"
        Exit Sub
    End If
    dpi = CLng(Val(cboDpi.Text))
    If dpi < 10 Or dpi > 2400 Then
        MsgBox "Razreshenie 10 dan 2400 gacha bo'lishi kerak.", vbExclamation, "Pechat eksport"
        Exit Sub
    End If

    pct = 0
    If chkBright.Value Then
        If Not IsNumeric(cboBright.Text) Then
            MsgBox "Ochroq qilish foizi raqam bo'lishi kerak.", vbExclamation, "Pechat eksport"
            Exit Sub
        End If
        pct = CLng(Val(cboBright.Text))
        If pct < 1 Or pct > 100 Then
            MsgBox "Ochroq qilish foizi 1 dan 100 gacha bo'lishi kerak.", vbExclamation, "Pechat eksport"
            Exit Sub
        End If
    End If

    If Len(txtFolder.Text) = 0 Or Len(Dir$(txtFolder.Text, vbDirectory)) = 0 Then
        MsgBox "Papka topilmadi: " & txtFolder.Text, vbExclamation, "Pechat eksport"
        Exit Sub
    End If

    SaveSettings
    btnExport.Enabled = False
    result = ExportPechat(optCMYK.Value, chkBright.Value, pct, SelectedType(), SelectedExt(), _
                          dpi, txtFolder.Text, optSel.Value)
    If Len(result) > 0 Then
        MsgBox "Tayyor! Fayl saqlandi:" & vbCrLf & result, vbInformation, "Pechat eksport"
        Unload Me
    Else
        btnExport.Enabled = True
    End If
End Sub
