'==============================================================================
'  frmExport - Pechat eksport oynasi
'  Bu kodni bo'sh UserForm (nomi: frmExport) ichiga to'liq qo'ying.
'  Barcha tugma va tanlovlar kod orqali avtomatik yaratiladi.
'==============================================================================
Option Explicit

' Ranglar
Private Const CLR_BG As Long = &H302D2D      ' to'q kulrang fon   RGB(45,45,48)
Private Const CLR_CTRL As Long = &H423E3E    ' maydonlar foni     RGB(62,62,66)
Private Const CLR_BTN As Long = &H4B4646     ' tugmalar foni      RGB(70,70,75)
Private Const CLR_TEXT As Long = &HFFFFFF    ' oq matn
Private Const FONT_SIZE As Single = 10

Private WithEvents cboColor As MSForms.ComboBox
Private WithEvents cboBright As MSForms.ComboBox
Private WithEvents cboType As MSForms.ComboBox
Private WithEvents cboFormat As MSForms.ComboBox
Private WithEvents cboDpi As MSForms.ComboBox
Private WithEvents btnFolder As MSForms.CommandButton
Private WithEvents btnExport As MSForms.CommandButton
Private WithEvents btnCancel As MSForms.CommandButton
Private txtFolder As MSForms.TextBox
Private lblInfo As MSForms.Label
Private lblPreview As MSForms.Label
Private lblNote As MSForms.Label
Private sel As ShapeRange
Private loading As Boolean

'------------------------------------------------------------------------------
'  Oyna yaratish
'------------------------------------------------------------------------------
Private Sub UserForm_Initialize()
    Const LX As Single = 14, CX As Single = 165, CW As Single = 200, ROW As Single = 30
    Dim t As Single
    loading = True
    Set sel = ActiveSelectionRange

    Me.Caption = "Pechatga eksport"
    Me.BackColor = CLR_BG
    Me.Width = 390
    Me.Height = 385

    t = 14
    AddLabel "Rang rejimi:", LX, t + 3, 145
    Set cboColor = AddCombo(CX, t, CW, True)
    cboColor.AddItem "CMYK": cboColor.AddItem "RGB"

    t = t + ROW
    AddLabel "Ranglarni ochroq qilish:", LX, t + 3, 150
    Set cboBright = AddCombo(CX, t, CW, True)
    Dim i As Long
    cboBright.AddItem "Yo'q"
    For i = 5 To 50 Step 5: cboBright.AddItem i & "%": Next i

    t = t + ROW
    AddLabel "Pechat turi:", LX, t + 3, 145
    Set cboType = AddCombo(CX, t, CW, True)
    cboType.AddItem "Banner": cboType.AddItem "Orakal": cboType.AddItem "Bekprint"

    t = t + ROW
    AddLabel "Format:", LX, t + 3, 145
    Set cboFormat = AddCombo(CX, t, CW, True)
    cboFormat.AddItem "TIF": cboFormat.AddItem "PNG": cboFormat.AddItem "JPG": cboFormat.AddItem "JPEG"

    t = t + ROW
    AddLabel "Razreshenie (dpi):", LX, t + 3, 145
    Set cboDpi = AddCombo(CX, t, CW, False)   ' o'zingiz ham yozsangiz bo'ladi
    cboDpi.AddItem "72": cboDpi.AddItem "100": cboDpi.AddItem "120": cboDpi.AddItem "150"
    cboDpi.AddItem "200": cboDpi.AddItem "300"

    t = t + ROW
    AddLabel "Papka:", LX, t + 3, 145
    Set txtFolder = Me.Controls.Add("Forms.TextBox.1")
    With txtFolder
        .Left = CX: .Top = t: .Width = CW - 32: .Height = 20
        .BackColor = CLR_CTRL: .ForeColor = CLR_TEXT
        .Font.Size = FONT_SIZE: .Font.Bold = True
    End With
    Set btnFolder = AddButton("...", CX + CW - 28, t, 28, 20)

    t = t + ROW + 4
    Set lblInfo = AddLabel("", LX, t, 350)
    t = t + 20
    Set lblPreview = AddLabel("", LX, t, 350)
    With lblPreview: .Height = 50: .ForeColor = RGB(120, 195, 255): .WordWrap = True: End With
    t = t + 52
    Set lblNote = AddLabel("", LX, t, 350)
    With lblNote: .Height = 30: .ForeColor = RGB(255, 175, 90): .WordWrap = True: End With

    t = t + 34
    Set btnExport = AddButton("Eksport", 175, t, 92, 30)
    btnExport.Default = True
    Set btnCancel = AddButton("Bekor qilish", 273, t, 92, 30)
    btnCancel.Cancel = True

    LoadSettings
    loading = False
    UpdateUI
End Sub

Private Function AddLabel(ByVal cap As String, ByVal l As Single, ByVal t As Single, _
        ByVal w As Single) As MSForms.Label
    Set AddLabel = Me.Controls.Add("Forms.Label.1")
    With AddLabel
        .Caption = cap: .Left = l: .Top = t: .Width = w: .Height = 18
        .BackStyle = fmBackStyleTransparent
        .ForeColor = CLR_TEXT
        .Font.Size = FONT_SIZE: .Font.Bold = True
    End With
End Function

Private Function AddCombo(ByVal l As Single, ByVal t As Single, ByVal w As Single, _
        ByVal listOnly As Boolean) As MSForms.ComboBox
    Set AddCombo = Me.Controls.Add("Forms.ComboBox.1")
    With AddCombo
        .Left = l: .Top = t: .Width = w: .Height = 20
        If listOnly Then .Style = fmStyleDropDownList
        .BackColor = CLR_CTRL: .ForeColor = CLR_TEXT
        .Font.Size = FONT_SIZE: .Font.Bold = True
    End With
End Function

Private Function AddButton(ByVal cap As String, ByVal l As Single, ByVal t As Single, _
        ByVal w As Single, ByVal h As Single) As MSForms.CommandButton
    Set AddButton = Me.Controls.Add("Forms.CommandButton.1")
    With AddButton
        .Caption = cap: .Left = l: .Top = t: .Width = w: .Height = h
        .BackColor = CLR_BTN: .ForeColor = CLR_TEXT
        .Font.Size = FONT_SIZE: .Font.Bold = True
    End With
End Function

' Ro'yxatdan matn bo'yicha tanlash (topilmasa defIndex)
Private Sub SelectItem(ByVal cbo As MSForms.ComboBox, ByVal txt As String, ByVal defIndex As Long)
    Dim i As Long
    For i = 0 To cbo.ListCount - 1
        If StrComp(cbo.List(i), txt, vbTextCompare) = 0 Then cbo.ListIndex = i: Exit Sub
    Next i
    cbo.ListIndex = defIndex
End Sub

'------------------------------------------------------------------------------
'  Sozlamalarni eslab qolish
'------------------------------------------------------------------------------
Private Sub LoadSettings()
    Dim folder As String
    SelectItem cboColor, GetSetting(APP_NAME, "Last", "Color", "CMYK"), 0
    SelectItem cboBright, GetSetting(APP_NAME, "Last", "Bright", "15%"), 3
    SelectItem cboType, GetSetting(APP_NAME, "Last", "Type", "Banner"), 0
    SelectItem cboFormat, GetSetting(APP_NAME, "Last", "Format", "TIF"), 0
    cboDpi.Text = GetSetting(APP_NAME, "Last", "Dpi", "150")

    folder = ActiveDocument.FilePath
    If Len(folder) = 0 Then folder = GetSetting(APP_NAME, "Last", "Folder", "")
    If Len(folder) = 0 Then folder = CreateObject("WScript.Shell").SpecialFolders("Desktop")
    txtFolder.Text = folder
End Sub

Private Sub SaveSettings()
    SaveSetting APP_NAME, "Last", "Color", cboColor.Text
    SaveSetting APP_NAME, "Last", "Bright", cboBright.Text
    SaveSetting APP_NAME, "Last", "Type", cboType.Text
    SaveSetting APP_NAME, "Last", "Format", cboFormat.Text
    SaveSetting APP_NAME, "Last", "Dpi", cboDpi.Text
    SaveSetting APP_NAME, "Last", "Folder", txtFolder.Text
End Sub

'------------------------------------------------------------------------------
'  Tanlovlar
'------------------------------------------------------------------------------
Private Function IsCMYK() As Boolean
    IsCMYK = (cboColor.Text = "CMYK")
End Function

Private Function BrightPct() As Long
    BrightPct = CLng(Val(cboBright.Text))   ' "Yo'q" -> 0, "15%" -> 15
End Function

Private Function NameFor(ByVal sr As ShapeRange) As String
    Dim x As Double, y As Double, w As Double, h As Double
    GetBoundsCm sr, x, y, w, h
    NameFor = BuildFileName(w, h, cboType.Text, IsCMYK(), cboFormat.Text)
End Function

Private Sub UpdateUI()
    Dim i As Long, txt As String, one As ShapeRange
    If loading Then Exit Sub

    ' PNG faqat RGB bo'ladi
    If cboFormat.Text = "PNG" Then
        If cboColor.Text <> "RGB" Then cboColor.ListIndex = 1
        cboColor.Enabled = False
        lblNote.Caption = "PNG formati CMYK ni qo'llamaydi - RGB bo'lib saqlanadi."
    Else
        cboColor.Enabled = True
        lblNote.Caption = ""
    End If

    If sel.Count = 0 Then
        lblInfo.Caption = "Hech narsa tanlanmagan - butun sahifa saqlanadi:"
        txt = NameFor(Nothing)
    Else
        If sel.Count = 1 Then
            lblInfo.Caption = "Tanlangan obyekt saqlanadi:"
        Else
            lblInfo.Caption = sel.Count & " ta obyekt - har biri alohida saqlanadi:"
        End If
        For i = 1 To sel.Count
            If i > 3 Then
                txt = txt & "... va yana " & (sel.Count - 3) & " ta"
                Exit For
            End If
            Set one = CreateShapeRange
            one.Add sel(i)
            txt = txt & NameFor(one) & vbCrLf
        Next i
    End If
    lblPreview.Caption = txt
End Sub

Private Sub cboColor_Change(): UpdateUI: End Sub
Private Sub cboType_Change(): UpdateUI: End Sub
Private Sub cboFormat_Change(): UpdateUI: End Sub

'------------------------------------------------------------------------------
'  Tugmalar
'------------------------------------------------------------------------------
Private Sub btnFolder_Click()
    Dim folder As String
    folder = CorelScriptTools.GetFolder(txtFolder.Text, "Eksport papkasini tanlang")
    If Len(folder) > 0 Then txtFolder.Text = folder
End Sub

Private Sub btnCancel_Click()
    Unload Me
End Sub

Private Sub btnExport_Click()
    Dim dpi As Long, result As String, n As Long, line As Variant

    If Not IsNumeric(cboDpi.Text) Then
        MsgBox "Razreshenie (dpi) raqam bo'lishi kerak.", vbExclamation, "Pechat eksport"
        Exit Sub
    End If
    dpi = CLng(Val(cboDpi.Text))
    If dpi < 10 Or dpi > 2400 Then
        MsgBox "Razreshenie 10 dan 2400 gacha bo'lishi kerak.", vbExclamation, "Pechat eksport"
        Exit Sub
    End If

    If Len(txtFolder.Text) = 0 Or Len(Dir$(txtFolder.Text, vbDirectory)) = 0 Then
        MsgBox "Papka topilmadi: " & txtFolder.Text, vbExclamation, "Pechat eksport"
        Exit Sub
    End If

    SaveSettings
    btnExport.Enabled = False
    lblNote.Caption = "Eksport qilinmoqda, kuting..."
    Me.Repaint

    result = ExportPechat(IsCMYK(), BrightPct(), cboType.Text, cboFormat.Text, dpi, txtFolder.Text)
    If Len(result) > 0 Then
        For Each line In Split(result, vbCrLf)
            If Len(line) > 0 Then n = n + 1
        Next line
        MsgBox "Tayyor! " & n & " ta fayl saqlandi:" & vbCrLf & vbCrLf & result, _
               vbInformation, "Pechat eksport"
        Unload Me
    Else
        btnExport.Enabled = True
        UpdateUI
    End If
End Sub
